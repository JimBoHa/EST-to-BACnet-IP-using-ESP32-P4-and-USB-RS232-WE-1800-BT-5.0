cmake_minimum_required(VERSION 3.20)
project(est3_native C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(BACNET_ROOT "${ROOT}/third_party/bacnet-stack/src")
set(CORE "${ROOT}/firmware/components/gateway_core")
set(PORT "${ROOT}/firmware/components/bacnet_gateway")
set(CJSON "$ENV{IDF_PATH}/components/json/cJSON")
if(NOT EXISTS "${CJSON}/cJSON.c")
    message(FATAL_ERROR "Set IDF_PATH to pinned ESP-IDF v5.5.5")
endif()
add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined -ffunction-sections -fdata-sections)
add_link_options(-fsanitize=address,undefined)
if(APPLE)
    add_link_options(-Wl,-dead_strip)
else()
    add_link_options(-Wl,--gc-sections)
endif()
include("${PORT}/sources.cmake")
add_library(gateway_core "${CORE}/gateway.c" "${CJSON}/cJSON.c")
target_include_directories(gateway_core PUBLIC "${CORE}" "${CJSON}")
target_compile_definitions(gateway_core PRIVATE CJSON_NESTING_LIMIT=16)
add_library(bacnet_gateway ${BACNET_SOURCES} "${PORT}/bacnet_port.c" "${PORT}/bacnet_gateway.c")
target_include_directories(bacnet_gateway PUBLIC "${BACNET_ROOT}" "${PORT}")
target_compile_definitions(bacnet_gateway PUBLIC BACDL_BIP BACAPP_ALL BACNET_PROTOCOL_REVISION=16 CONFIG_BACNET_BASIC_OBJECT_BINARY_INPUT MAX_APDU=1476 MAX_TSM_TRANSACTIONS=4 PRINT_ENABLED=0 BACNET_VENDOR_ID=65535)
target_link_libraries(bacnet_gateway PUBLIC gateway_core)
add_executable(test_core test_core.c)
target_link_libraries(test_core gateway_core)
add_test(NAME core COMMAND test_core "${ROOT}/evidence/sim-registry.json")
add_executable(test_registry_scaling test_registry_scaling.c)
target_link_libraries(test_registry_scaling gateway_core)
add_test(NAME registry_scaling COMMAND test_registry_scaling)
add_executable(test_serial_diagnostics test_serial_diagnostics.c "${ROOT}/firmware/main/serial_diagnostics.c")
target_include_directories(test_serial_diagnostics PRIVATE "${ROOT}/firmware/main")
add_test(NAME serial_diagnostics COMMAND test_serial_diagnostics)
add_library(printer "${ROOT}/firmware/components/printer/printer.c" "${ROOT}/firmware/components/printer/printer_event.c")
target_include_directories(printer PUBLIC "${ROOT}/firmware/components/printer")
add_library(printer_observer "${ROOT}/firmware/components/printer_observer/printer_observer.c")
target_include_directories(printer_observer PUBLIC "${ROOT}/firmware/components/printer_observer")
target_link_libraries(printer_observer PUBLIC gateway_core printer)
add_executable(test_printer test_printer.c)
target_link_libraries(test_printer printer)
add_test(NAME printer COMMAND test_printer "${ROOT}/fixtures/printer-revision-sanitized.txt")
add_executable(test_printer_events test_printer_events.c)
target_link_libraries(test_printer_events printer_observer)
add_test(NAME printer_events COMMAND test_printer_events "${ROOT}/fixtures/printer-trouble-sanitized.txt")
add_executable(printer_bacnet "${ROOT}/simulator/printer_bacnet.c")
target_link_libraries(printer_bacnet printer_observer bacnet_gateway)
add_executable(sim_bacnet "${ROOT}/simulator/native_bacnet.c")
target_link_libraries(sim_bacnet bacnet_gateway)
add_executable(audit_registry audit_registry.c)
target_link_libraries(audit_registry bacnet_gateway)
